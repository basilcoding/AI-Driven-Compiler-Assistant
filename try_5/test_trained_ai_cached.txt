_CANONICAL_BY_STRIPPED = {p.lstrip('-').lower(): p for p in PASS_DESCRIPTIONS}

def canonicalize_pass(name: str) -> str:
    key = name.strip().lstrip('-').lower()
    return _CANONICAL_BY_STRIPPED.get(key, name) 

def format_spellbook_by_category() -> str:
    buckets = {}
    for p, desc in PASS_DESCRIPTIONS.items():
        clean_desc = desc.rstrip('.')
        buckets.setdefault(clean_desc, []).append(p)
        
    return "\n".join(
        f"  - {desc}: {', '.join(sorted(names))}"
        for desc, names in sorted(buckets.items())
    )

GRAND_SPELLBOOK_BLOCK = format_spellbook_by_category()

rules = (
    "Rules:\n"
    "- Pick however many passes are actually useful — no bonus for more, no penalty for fewer.\n"
    "- End with your chosen passes as a JSON array of strings inside <answer>...</answer>, nothing else inside the tags.\n"
)

SYSTEM_PROMPT = (
    "You are an LLVM Compiler RL Agent. Your objective is to discover a custom sequence of optimization passes that will shrink the instruction count of the given IR even further than the standard '-Oz' optimization level.\n\n"
    "The only passes you can use along with their descriptions are:\n"
    f"{GRAND_SPELLBOOK_BLOCK}\n\n"
    f"{rules}"
)

# ==========================================
# 3. LOAD MODEL 
# ==========================================
model, tokenizer = FastLanguageModel.from_pretrained(
    model_name = MODEL_PATH,
    max_seq_length = MAX_SEQ_LENGTH,
    load_in_4bit = LOAD_IN_4BIT,
)
FastLanguageModel.for_inference(model)

# THIS IS THE MISSING LINE:
streamer = TextStreamer(tokenizer, skip_prompt=True, skip_special_tokens=True)

print("\n" + "="*60)
print("Model loaded! Ready for inputs.")

while True:
    print("\n" + "="*60)
    print("Paste your AutoPhase dictionary (or type 'quit' to exit).")
    print("IMPORTANT: When you are done pasting, press Enter on a BLANK LINE to start generating:")
    
    # Read multi-line input until a blank line
    lines = []
    while True:
        line = input()
        if not line: # Break the loop if the line is completely blank
            break
        lines.append(line)
        
    user_input = "\n".join(lines).strip()
    if user_input.lower() == "quit":
        break
    if not user_input:
        continue

    # Format the prompt
    USER_PROMPT = f"\nAutoPhase: {user_input}\n\nRemember: conclude your response with exactly one <answer>...</answer> block containing a JSON array of pass names as strings."
    
    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": USER_PROMPT},
    ]

    prompt = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    inputs = tokenizer(prompt, return_tensors="pt", add_special_tokens=True).to(model.device)

    print("\nGenerating...")
    with torch.no_grad():
        output_ids = model.generate(
            **inputs,
            streamer = streamer, # Now this variable actually exists!
            max_new_tokens = 3000,
            temperature = 0.6,
            top_p = 0.95,
            do_sample = True,
            eos_token_id = tokenizer.eos_token_id,
            pad_token_id = tokenizer.eos_token_id,
        )

    # Extract answer
    full_text = tokenizer.decode(output_ids[0][inputs.input_ids.shape[1]:], skip_special_tokens=True)
    m = re.search(r"<answer>\s*(.*?)\s*</answer>", full_text, re.DOTALL)
    
    print("\n" + "-" * 30)
    if m:
        passes = [p.strip() for p in m.group(1).split(",") if p.strip()]
        print(f"EXTRACTED PASS SEQUENCE ({len(passes)} passes):\n{','.join(passes)}")
    else:
        print("WARNING: no <answer>...</answer> block found in output!")
